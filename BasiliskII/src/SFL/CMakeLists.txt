cmake_minimum_required(VERSION 3.16)
if(CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR)
	project(BASILISKIISFL C CXX)
else()
	unset(CMAKE_C_STANDARD)
endif()
option(SFLMONITOR "Build the emulator monitor server" OFF)
set(CMAKE_CXX_STANDARD 98)
set(CMAKE_CXX_STANDARD_REQUIRED OFF)
set(BASILISKIISOURCE "${CMAKE_CURRENT_SOURCE_DIR}/..")
include(SFL.cmake)
set(UAEDIRECTORY "${BASILISKIISOURCE}/uae_cpu")
set(BASILISKIIGENERATED "${CMAKE_CURRENT_BINARY_DIR}/uae_gen")
file(MAKE_DIRECTORY "${BASILISKIIGENERATED}")
if(CMAKE_HOST_WIN32 AND CMAKE_SYSTEM_NAME MATCHES RetroCarbon)
	add_compile_definitions(__STRINGS__)
	add_compile_options("SHELL:-include bsdstrings.h")
endif()
set(BASILISKIIUAEGENERATEDSOURCES
	"${BASILISKIIGENERATED}/cpuemu.cpp" "${BASILISKIIGENERATED}/cpustbl.cpp" "${BASILISKIIGENERATED}/cputbl.h"
	"${BASILISKIIGENERATED}/cpuemu_nf.cpp" "${BASILISKIIGENERATED}/cpustbl_nf.cpp")
set(BASILISKIIHOSTTOOLS "" CACHE PATH "Directory holding a native SFL host-tools build, for cross builds")
if(NOT BASILISKIIHOSTTOOLS AND MSVC AND CMAKE_C_COMPILER_ARCHITECTURE_ID STREQUAL "ARM64"
		AND NOT CMAKE_HOST_SYSTEM_PROCESSOR MATCHES "ARM64")
	message(FATAL_ERROR "Cross compiling for ARM64: set BASILISKIIHOSTTOOLS to a native SFL build's host-tools directory")
endif()
if(BASILISKIIHOSTTOOLS)
	add_custom_command(
		OUTPUT "${BASILISKIIGENERATED}/cpudefs.cpp"
		COMMAND ${CMAKE_COMMAND}
		-DMACEMU_REDIRECT_TOOL=${BASILISKIIHOSTTOOLS}/sfl_build68k${CMAKE_HOST_EXECUTABLE_SUFFIX}
		"-DMACEMU_REDIRECT_ARGS=\"${UAEDIRECTORY}/table68k\""
		-DMACEMU_REDIRECT_OUTPUT=${BASILISKIIGENERATED}/cpudefs.cpp
		-P "${BASILISKIISOURCE}/../../cmake/run_redirect.cmake"
		DEPENDS "${UAEDIRECTORY}/table68k"
		COMMENT "Generating cpudefs.cpp"
		VERBATIM)
	add_custom_command(
		OUTPUT ${BASILISKIIUAEGENERATEDSOURCES}
		COMMAND ${BASILISKIIHOSTTOOLS}/sfl_gencpu${CMAKE_HOST_EXECUTABLE_SUFFIX}
		DEPENDS "${BASILISKIIGENERATED}/cpudefs.cpp"
		WORKING_DIRECTORY "${BASILISKIIGENERATED}"
		COMMENT "Running gencpu"
		VERBATIM)
	add_custom_target(sfl_uae_tables DEPENDS ${BASILISKIIUAEGENERATEDSOURCES})
else()
	include(ExternalProject)
	ExternalProject_Add(sfl_uae_tables
		SOURCE_DIR "${SFLDIRECTORY}/HostTools"
		BINARY_DIR "${CMAKE_CURRENT_BINARY_DIR}/host-tools"
		CMAKE_ARGS "-DBASILISKIIGENERATED=${BASILISKIIGENERATED}" "-DCMAKE_BUILD_TYPE=Release"
			"-DCMAKE_MAKE_PROGRAM=${CMAKE_MAKE_PROGRAM}"
		BUILD_COMMAND "${CMAKE_COMMAND}" --build <BINARY_DIR> --config Release --target sfl_uae_tables
		BUILD_ALWAYS TRUE
		BUILD_BYPRODUCTS "${BASILISKIIGENERATED}/cpudefs.cpp" ${BASILISKIIUAEGENERATEDSOURCES}
		INSTALL_COMMAND "")
endif()
set_source_files_properties("${BASILISKIIGENERATED}/cpudefs.cpp" ${BASILISKIIUAEGENERATEDSOURCES} PROPERTIES GENERATED TRUE)

set(BASILISKIICORE
	"${BASILISKIISOURCE}/main.cpp"
	"${BASILISKIISOURCE}/prefs.cpp"
	"${BASILISKIISOURCE}/prefs_items.cpp"
	"${BASILISKIISOURCE}/rom_patches.cpp"
	"${BASILISKIISOURCE}/slot_rom.cpp"
	"${BASILISKIISOURCE}/rsrc_patches.cpp"
	"${BASILISKIISOURCE}/emul_op.cpp"
	"${BASILISKIISOURCE}/macos_util.cpp"
	"${BASILISKIISOURCE}/xpram.cpp"
	"${BASILISKIISOURCE}/timer.cpp"
	"${BASILISKIISOURCE}/adb.cpp"
	"${BASILISKIISOURCE}/serial.cpp"
	"${BASILISKIISOURCE}/ether.cpp"
	"${BASILISKIISOURCE}/sony.cpp"
	"${BASILISKIISOURCE}/disk.cpp"
	"${BASILISKIISOURCE}/qcow.cpp"
	"${BASILISKIISOURCE}/chd.cpp"
	"${BASILISKIISOURCE}/imageio.cpp"
	"${BASILISKIISOURCE}/cdrom.cpp"
	"${BASILISKIISOURCE}/scsi.cpp"
	"${BASILISKIISOURCE}/video.cpp"
	"${BASILISKIISOURCE}/audio.cpp"
	"${BASILISKIISOURCE}/extfs.cpp"
	"${BASILISKIISOURCE}/user_strings.cpp"
	"${BASILISKIISOURCE}/printing.cpp"
	"${BASILISKIISOURCE}/pictpdf.c"
	"${BASILISKIISOURCE}/pdfgen.c"
	"${BASILISKIISOURCE}/joymanager.cpp"
	"${BASILISKIISOURCE}/bincue.cpp"
)


set(BASILISKIIUAE
	"${UAEDIRECTORY}/basilisk_glue.cpp"
	"${UAEDIRECTORY}/memory.cpp"
	"${UAEDIRECTORY}/newcpu.cpp"
	"${UAEDIRECTORY}/readcpu.cpp"
	"${UAEDIRECTORY}/fpu/fpu_ieee.cpp"
	"${BASILISKIIGENERATED}/cpudefs.cpp"
	"${BASILISKIIGENERATED}/cpuemu.cpp"
	"${BASILISKIIGENERATED}/cpustbl.cpp"
	"${BASILISKIIGENERATED}/cpuemu_nf.cpp"
	"${BASILISKIIGENERATED}/cpustbl_nf.cpp"
)

add_library(BASILISKIISFLCORE OBJECT ${BASILISKIICORE} ${BASILISKIIUAE} ${SFLPLATFORMSOURCES}
	"${SFLDIRECTORY}/video_sfl.cpp")
add_dependencies(BASILISKIISFLCORE sfl_uae_tables)
if(NOT CMAKE_SYSTEM_NAME MATCHES RetroCarbon)
	target_sources(BASILISKIISFLCORE PRIVATE "${BASILISKIISOURCE}/CrossPlatform/sigsegv.cpp")
	target_compile_definitions(BASILISKIISFLCORE PRIVATE SFLFAULTHANDLER=1)
	if(WIN32)
		set_source_files_properties("${BASILISKIISOURCE}/CrossPlatform/sigsegv.cpp" PROPERTIES COMPILE_DEFINITIONS HAVE_WIN32_VM=1)
	endif()
endif()
target_include_directories(BASILISKIISFLCORE PRIVATE "${SFLDIRECTORY}" "${BASILISKIISOURCE}/include"
	"${BASILISKIISOURCE}/CrossPlatform" "${UAEDIRECTORY}" "${BASILISKIIGENERATED}" "${SFLLIBRARIES}")
sfl_configure_core(BASILISKIISFLCORE)
sfl_add_application(BASILISKIISFL B2EM BASILISKIISFLCORE)

if(APPLE)
	set_source_files_properties("${BASILISKIISOURCE}/pdfgen.c" PROPERTIES COMPILE_OPTIONS -fno-builtin-strrchr)
endif()
